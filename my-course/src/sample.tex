\begin{abstract}
Demonstration of every v1 pilot: \texttt{ximera-hint}, \texttt{ximera-word-choice}, \texttt{ximera-multiple-choice}, \texttt{ximera-select-all}, \texttt{ximera-free-response}, and the flagship \texttt{ximera-answer}. Feedback works out of the box via the kernel.
\end{abstract}

\begin{problem}
  My favorite number is $y = \answer{17}$.

  \begin{feedback}[attempt]
    You made a first attempt!
  \end{feedback}

  \begin{feedback}[correct]
    I have always loved the number 17.
  \end{feedback}

  \begin{hint}
    It's the smallest prime that is neither a Fermat prime nor a Mersenne prime.
  \end{hint}

  \begin{hint}
    It's the number of syllables in a haiku.
  \end{hint}
\end{problem}

\begin{problem}
Here is a problem with a subproblem.
  First, $1+1 = \answer{2}$.

\begin{problem}
And then $1+1+1 = \answer{3}$.
\end{problem}

\end{problem}

\begin{problem}
 A common approximation to $\sqrt{2}$, correct to two decimal places, is
  \[
    \sqrt{2} \approx \answer[format=float,tolerance=0.01]{1.414}.
  \]

But exactly,
  \[
    \sqrt{2} = \sqrt{\answer{2}}.
  \]

Or another thing would be
  \[
    \sqrt{1+1} = \answer{\sqrt{2}}.
  \]

\end{problem}

\begin{problem}
  The capital of Ohio is \wordChoice{%
    \choice{Cleveland and $\sqrt{2}$}%
    \choice[correct]{Columbus with $\cos 45$}%
    \choice{Cincinnati}%
  }.
\end{problem}

\begin{problem}
  Which of these is prime?
  \begin{multipleChoice}
    \choice{4}
    \choice{6}
    \choice[correct]{$1+\sqrt{256}$}
    \choice{$3^2$}
  \end{multipleChoice}
\end{problem}

\begin{problem}
  Which of these are even numbers?
  \begin{selectAll}
    \choice[correct]{2}
    \choice{3}
    \choice[correct]{4}
    \choice{5}
    \choice[correct]{6}
  \end{selectAll}
\end{problem}

\begin{problem}
  In your own words, describe why the sum of two odd numbers is always even.
  \begin{freeResponse}
    Write your explanation here.
  \end{freeResponse}
\end{problem}

\begin{dialogue}
  \item[Alice] I've been thinking about $\sqrt{2}$.
  \item[Bob] What about it?
  \item[Alice] It's irrational, but I can approximate it: $\sqrt{2} \approx 1.414$.
  \item[Bob] Prove it's irrational!
  \item[Alice] Suppose $\sqrt{2} = p/q$ in lowest terms. Then $2q^2 = p^2$, so $p$ is even, so $p^2$ is divisible by 4, so $q^2$ is even too --- contradicting lowest terms.
\end{dialogue}
